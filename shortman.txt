User prompt: 2.Write a Java program to implement a Swing.
Response: Aim
To write a simple Java program to create a graphical user interface (GUI) using basic Swing components (JFrame,
JButton, and JLabel).
Algorithm
.. Import Swing packages: javax.swing.* and java.awt.event.*.
2. Create a top-level window using JFrame.
3. Create a JButton (click button) and a JLabel (text display).
4. Set the position and size of both components using setBounds().
5. Attach an ActionListener to the button to update the label's text when clicked.
6. Add the components to the frame, set layout to null, set window size, and make it visible using setVisible(true).
Java Program
import javax.swing.*;
import java.awt.event.*;
public class SwingDemo {
    public static void main(String[] args) {
        // 1. Create Frame (window)
        JFrame f = new JFrame("Swing Example");
        // 2. Create Label and Button
        JLabel label = new JLabel("Click the button");
        label.setBounds(50, 50, 200, 30);
        JButton btn = new JButton("Click Me");
        btn.setBounds(50, 100, 100, 30);
        // 3. Add Click Action
        btn.addActionListener(new ActionListener() {
            public void actionPerformed(ActionEvent e) {
                label.setText("Hello, Swing!");
            }
        });
        // 4. Add components to Frame
        f.add(label);
        f.add(btn);
        // 5. Window settings
        f.setSize(300, 250);
        f.setLayout(null);
        f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        f.setVisible(true);
    }
}

public class FactorialApplet extends Applet implements ActionListener {
    TextField txtNum;
    Button btnCompute;
    Label lblResult;
    public void init() {
        // Create components
        txtNum = new TextField(10);
        btnCompute = new Button("Compute");
        lblResult = new Label("Result: ");
        // Add components to Applet
        add(new Label("Enter Number:"));
        add(txtNum);
        add(btnCompute);
        add(lblResult);
        // Register button listener
        btnCompute.addActionListener(this);
    }
    public void actionPerformed(ActionEvent e) {
        int n = Integer.parseInt(txtNum.getText());
        long fact = 1;
        for (int i = 1; i <= n; i++) {
            fact *= i;
        }
        lblResult.setText("Factorial: " + fact);
    }
}

Algorithm

Java Program
import java.applet.Applet;
import java.awt.*;
import java.awt.event.*;
/* <applet code="FactApplet.class" width="250" height="150"></applet> */
public class FactApplet extends Applet implements ActionListener {
    TextField t = new TextField(5);
    Button b = new Button("Compute");
    int ans = 1;
    public void init() {
        add(t);
        add(b);
        b.addActionListener(this);
    }
    public void actionPerformed(ActionEvent e) {
        int n = Integer.parseInt(t.getText());
        ans = 1;
        for (int i = 1; i <= n; i++) ans *= i;
        repaint();
    }
    public void paint(Graphics g) {
        g.drawString("Factorial: " + ans, 20, 80);
    }
}

import java.io.IOException;
import java.io.PrintWriter;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
public class HelloServlet extends HttpServlet {
    public void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        // 1. Set response content type
        response.setContentType("text/html");
        // 2. Get output stream
        PrintWriter out = response.getWriter();
        // 3. Print message to browser
        out.println("<h1>Hello World</h1>");
    }
}
