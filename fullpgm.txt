1. JDBC Database Connectivity
Code:
```java
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.ResultSet;
import java.sql.Statement;
public class JDBCDemo {
    public static void main(String[] args) {
        String url = "jdbc:mysql://localhost:3306/testdb";
        String user = "root";
        String password = "";
        try {
            Class.forName("com.mysql.cj.jdbc.Driver");
            Connection con = DriverManager.getConnection(url, user, password);
            System.out.println("Connected to database successfully!");
            Statement stmt = con.createStatement();
            ResultSet rs = stmt.executeQuery("SELECT 1");
            if (rs.next()) {
                System.out.println("Query executed successfully.");
            }
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
```
Expected Output:
```
Connected to database successfully!
Query executed successfully.
```

```java
    public void keyTyped(KeyEvent e) { }

        new KeyEventDemo();
    }
```

```java
import java.awt.Color;

    }
    public void mouseEntered(MouseEvent me) {
        msg = "Mouse Entered Window";
        repaint();
    }
    public void mouseExited(MouseEvent me) {
        msg = "Mouse Exited Window";
        repaint();
    }
    public void mousePressed(MouseEvent me) {
        msg = "Mouse Pressed";
        repaint();
    }
    public void mouseReleased(MouseEvent me) {
        msg = "Mouse Released";
        repaint();
    }
    public void paint(Graphics g) {
        g.setColor(Color.BLUE);
        g.drawString(msg, 50, 100);
```

```java

    public void run() {
        for (int i = 1; i <= 3; i++) {
            System.out.println("Thread 1 executing step " + i);
            try {
                Thread.sleep(500);
            } catch (InterruptedException e) {
                System.out.println(e);
            }
        }
    }
}
class TaskTwo implements Runnable {
    public void run() {
        for (int i = 1; i <= 3; i++) {
            System.out.println("Thread 2 executing step " + i);
            try {
                Thread.sleep(500);
            } catch (InterruptedException e) {
                System.out.println(e);
            }
        }
    }
}
public class MultiThreadingDemo {
    public static void main(String[] args) {
        TaskOne t1 = new TaskOne();
        Thread t2 = new Thread(new TaskTwo());
        t1.start();
        t2.start();
    }
}
```
Expected Output:
```
Thread 1 executing step 1
Thread 2 executing step 1
Thread 1 executing step 2
Thread 2 executing step 2
Thread 1 executing step 3
Thread 2 executing step 3
```

14. Applet for Factorial Calculation
Code:
```java
import java.applet.Applet;
import java.awt.Button;
import java.awt.Label;
import java.awt.TextField;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;
/
<applet code="FactorialApplet.class" width="350" height="200">
</applet>
/
public class FactorialApplet extends Applet implements ActionListener {
    Label lNum, lResult;
    TextField tNum, tResult;
    Button bCompute;
    public void init() {
        lNum = new Label("Enter Integer:");
        tNum = new TextField(10);
        bCompute = new Button("Compute");
        lResult = new Label("Factorial:");
        tResult = new TextField(15);
        tResult.setEditable(false);
        add(lNum);
        add(tNum);
        add(bCompute);
        add(lResult);
        add(tResult);
        bCompute.addActionListener(this);
    }
    public void actionPerformed(ActionEvent ae) {
        try {
            int n = Integer.parseInt(tNum.getText().trim());
            long fact = 1;
            for (int i = 1; i <= n; i++) {
                fact = i;
            }
            tResult.setText(String.valueOf(fact));
        } catch (NumberFormatException ex) {
            tResult.setText("Invalid Number");
        }
    }
}
```
Expected Output:
![FactorialApplet Output]
(Input 5 → Output shows "120")
